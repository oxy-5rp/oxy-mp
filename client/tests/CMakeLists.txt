# Тесты счёта промежуточных состояний.
#
# Единственная часть клиента, которую можно проверить без запущенной игры: она
# не зовёт нативов и ничего не знает ни про машины, ни про персонажей — только
# считает, где между двумя снимками мы находимся сейчас. Именно в этом счёте и
# жила ошибка, из-за которой чужие игроки стояли повёрнутыми не туда.
#
# Второе, что проверяется без игры, — чтение свёртка ресурса. Игра ему тоже не
# нужна: свёрток собирается прямо в проверке, кладётся во временный каталог и
# читается обратно.
#
# Третье — разбор того, кто в этом кадре распоряжается телом чужого игрока. Он
# нарочно отделён от нативов и потому проверяется здесь: решение это одно из
# самых запутанных в клиенте, а цена ошибки в нём — замерший столбом человек,
# по которому нельзя понять, какой из шести признаков залип.
#
# Четвёртое — укладка доводов натива. Игры она тоже не требует: это чистая
# раскладка памяти. Проверяется потому, что ошибка в ней не видна ниоткуда —
# натив молча получает не то число и молча отвечает не то. Так и вышло с местом
# водителя: -1 уезжало в игру как 255.
add_executable(oxymp_client_tests
    bundle_store_tests.cpp
    interpolation_tests.cpp
    native_context_tests.cpp
    puppet_duty_tests.cpp
    ../src/bundle_store.cpp
    ../src/interpolation.cpp
    ../src/game/puppet_duty.cpp
)

# Заголовок счёта лежит там же, где и остальные заголовки клиентского ядра:
# им пользуется не только этот набор, но и connection.hpp — оценка отставания
# хранится у каждого чужого игрока и у каждой машины.
target_include_directories(oxymp_client_tests PRIVATE ../include)

target_link_libraries(oxymp_client_tests PRIVATE
    oxymp::shared
    oxymp::options
    spdlog::spdlog
    Catch2::Catch2WithMain
)

include(Catch)
catch_discover_tests(oxymp_client_tests)
